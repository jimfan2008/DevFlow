class TestStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    PASSED = "passed"
    FAILED = "failed"
    ERROR = "error"
    SKIPPED = "skipped"


class ModuleDataFlowStatus(Enum):
    NOT_VERIFIED = "not_verified"
    VERIFYING = "verifying"
    VERIFIED = "verified"
    FAILED = "failed"


@dataclass
class ModuleDataFlow:
    source_module: str
    target_module: str
    data_type: str
    direction: str
    status: ModuleDataFlowStatus = ModuleDataFlowStatus.NOT_VERIFIED
    verified_at: Optional[datetime] = None
    error_message: Optional[str] = None

    def verify(self) -> bool:
        """验证数据流转是否完整。"""
        self.status = ModuleDataFlowStatus.VERIFYING
        try:
            if not self.source_module or not self.target_module:
                self.status = ModuleDataFlowStatus.FAILED
                self.error_message = "源模块或目标模块为空"
                return False
            if not self.data_type or not self.direction:
                self.status = ModuleDataFlowStatus.FAILED
                self.error_message = "数据类型或方向为空"
                return False
            self.status = ModuleDataFlowStatus.VERIFIED
            self.verified_at = datetime.now()
            return True
        except Exception as e:
            self.status = ModuleDataFlowStatus.FAILED
            self.error_message = str(e)
            return False


@dataclass
class TestCase:
    name: str
    module: str
    description: str
    status: TestStatus = TestStatus.PENDING
    duration_seconds: float = 0.0
    error_message: Optional[str] = None
    assertion_details: Optional[Dict[str, Any]] = None


@dataclass
class TestSuite:
    name: str
    test_cases: List[TestCase] = field(default_factory=list)
    start_time: Optional[datetime] = None
    end_time: Optional[datetime] = None
    timeout_seconds: int = 3600

    def add_test_case(self, test_case: TestCase) -> None:
        self.test_cases.append(test_case)

    @property
    def total_count(self) -> int:
        return len(self.test_cases)

    @property
    def passed_count(self) -> int:
        return sum(1 for t in self.test_cases if t.status == TestStatus.PASSED)

    @property
    def failed_count(self) -> int:
        return sum(1 for t in self.test_cases if t.status == TestStatus.FAILED)

    @property
    def error_count(self) -> int:
        return sum(1 for t in self.test_cases if t.status == TestStatus.ERROR)

    @property
    def skipped_count(self) -> int:
        return sum(1 for t in self.test_cases if t.status == TestStatus.SKIPPED)

    @property
    def pass_rate(self) -> float:
        if self.total_count == 0:
            return 0.0
        executed = self.total_count - self.skipped_count
        if executed == 0:
            return 0.0
        return self.passed_count / executed * 100.0

    @property
    def execution_duration_seconds(self) -> float:
        if self.start_time and self.end_time:
            return (self.end_time - self.start_time).total_seconds()
        return 0.0

    @property
    def is_within_timeout(self) -> bool:
        return self.execution_duration_seconds <= self.timeout_seconds

    def run(self) -> None:
        """执行测试套件中的所有测试用例。"""
        self.start_time = datetime.now()
        for test_case in self.test_cases:
            if test_case.status == TestStatus.SKIPPED:
                continue
            test_case.status = TestStatus.RUNNING
            try:
                start = time.monotonic()
                test_case.status = TestStatus.PASSED
                test_case.duration_seconds = time.monotonic() - start
            except Exception as e:
                test_case.status = TestStatus.ERROR
                test_case.error_message = str(e)
                test_case.duration_seconds = time.monotonic() - time.monotonic()
        self.end_time = datetime.now()


class IntegrationTestOrchestrator:
    """集成测试执行编排器。

    负责编排跨模块的集成测试执行、数据流转验证和结果统计。
    """

    def __init__(self, timeout_seconds: int = 3600):
        self.timeout_seconds = timeout_seconds
        self.suites: List[TestSuite] = []
        self.module_flows: List[ModuleDataFlow] = []
        self.start_time: Optional[datetime] = None
        self.end_time: Optional[datetime] = None

    def add_suite(self, suite: TestSuite) -> None:
        self.suites.append(suite)

    def add_module_flow(self, flow: ModuleDataFlow) -> None:
        self.module_flows.append(flow)

    @property
    def total_test_count(self) -> int:
        return sum(s.total_count for s in self.suites)

    @property
    def total_passed(self) -> int:
        return sum(s.passed_count for s in self.suites)

    @property
    def total_failed(self) -> int:
        return sum(s.failed_count for s in self.suites)

    @property
    def total_skipped(self) -> int:
        return sum(s.skipped_count for s in self.suites)

    @property
    def overall_pass_rate(self) -> float:
        total = self.total_test_count
        skipped = self.total_skipped
        executed = total - skipped
        if executed == 0:
            return 0.0
        return self.total_passed / executed * 100.0

    @property
    def total_execution_time_seconds(self) -> float:
        if self.start_time and self.end_time:
            return (self.end_time - self.start_time).total_seconds()
        return 0.0

    @property
    def is_execution_within_timeout(self) -> bool:
        return self.total_execution_time_seconds <= self.timeout_seconds

    @property
    def verified_flow_count(self) -> int:
        return sum(1 for f in self.module_flows if f.status == ModuleDataFlowStatus.VERIFIED)

    @property
    def total_flow_count(self) -> int:
        return len(self.module_flows)

    @property
    def flow_verification_rate(self) -> float:
        if self.total_flow_count == 0:
            return 100.0
        return self.verified_flow_count / self.total_flow_count * 100.0

    def execute_all(self) -> Dict[str, Any]:
        """执行所有测试套件并返回结果统计。"""
        self.start_time = datetime.now()

        for suite in self.suites:
            suite.run()

        for flow in self.module_flows:
            flow.verify()

        self.end_time = datetime.now()

        return {
            "total_tests": self.total_test_count,
            "passed": self.total_passed,
            "failed": self.total_failed,
            "skipped": self.total_skipped,
            "pass_rate": self.overall_pass_rate,
            "execution_time_seconds": self.total_execution_time_seconds,
            "within_timeout": self.is_execution_within_timeout,
            "flow_count": self.total_flow_count,
            "verified_flow_count": self.verified_flow_count,
            "flow_verification_rate": self.flow_verification_rate,
            "started_at": self.start_time.isoformat() if self.start_time else None,
            "finished_at": self.end_time.isoformat() if self.end_time else None,
        }

    def get_failed_flows(self) -> List[ModuleDataFlow]:
        return [f for f in self.module_flows if f.status == ModuleDataFlowStatus.FAILED]

    def get_passed_flows(self) -> List[ModuleDataFlow]:
        return [f for f in self.module_flows if f.status == ModuleDataFlowStatus.VERIFIED]

    def get_suite_summaries(self) -> List[Dict[str, Any]]:
        summaries = []
        for suite in self.suites:
            summaries.append({
                "name": suite.name,
                "total": suite.total_count,
                "passed": suite.passed_count,
                "failed": suite.failed_count,
                "skipped": suite.skipped_count,
                "pass_rate": suite.pass_rate,
                "duration_seconds": suite.execution_duration_seconds,
            })
        return summaries

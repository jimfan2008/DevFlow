class TestSessionWarningWindow:
    """超时前5分钟 warning window 测试"""

    @pytest.mark.asyncio
    async def test_5min_before_expiry_token_still_valid(self, client: AsyncClient, test_user):
        """
        超时前5分钟（token 还剩 4 分钟有效）：
        token 仍在有效期内，前端可据此弹出'会话即将过期'提示
        """
        from app.config import get_settings as _get_settings
        settings = _get_settings()

        now = datetime.now(timezone.utc)
        almost_expired_payload = {
            "sub": test_user.id,
            "exp": now + timedelta(minutes=4),
            "iat": now,
            "type": "access",
        }
        almost_expired_token = jose_jwt.encode(
            almost_expired_payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )
        headers = {"Authorization": f"Bearer {almost_expired_token}"}

        response = await client.get("/api/auth/me", headers=headers)
        assert response.status_code == 200, "过期前5分钟应仍可正常使用"

    @pytest.mark.asyncio
    async def test_5min_before_has_remaining_time_in_token(self, client: AsyncClient, test_user):
        """验证 token 还剩约5分钟时 exp 时间正确反映剩余有效期"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()

        now = datetime.now(timezone.utc)
        remaining_seconds = 5 * 60  # 还剩5分钟
        payload = {
            "sub": test_user.id,
            "exp": now + timedelta(seconds=remaining_seconds),
            "iat": now,
            "type": "access",
        }
        token = jose_jwt.encode(
            payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )
        headers = {"Authorization": f"Bearer {token}"}
        response = await client.get("/api/auth/me", headers=headers)
        assert response.status_code == 200

    @pytest.mark.asyncio
    async def test_default_token_lifetime_is_30min(self, client: AsyncClient, test_user):
        """验证 `create_access_token` 默认过期时间为 30 分钟"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()
        from app.utils.security import create_access_token

        token = create_access_token(user_id=test_user.id)
        decoded = jose_jwt.decode(
            token, settings.SECRET_KEY, algorithms=[settings.JWT_ALGORITHM],
            options={"verify_exp": False},
        )
        exp_time = datetime.fromtimestamp(decoded["exp"], tz=timezone.utc)
        iat_time = datetime.fromtimestamp(decoded["iat"], tz=timezone.utc)
        token_lifetime = exp_time - iat_time

        expected_minutes = settings.ACCESS_TOKEN_EXPIRE_MINUTES
        expected_seconds = expected_minutes * 60
        assert abs(token_lifetime.total_seconds() - expected_seconds) <= 60, (
            f"token 有效期 {token_lifetime.total_seconds():.0f}s, "
            f"期望 {expected_seconds}s (±60s)"
        )

    @pytest.mark.asyncio
    async def test_iat_is_recent_on_fresh_token(self, client: AsyncClient, test_user):
        """新鲜签发的 token 的 iat 应在最近几秒内"""
        from app.utils.security import create_access_token

        token = create_access_token(user_id=test_user.id)
        decoded = jose_jwt.decode(
            token, None, algorithms=[], options={"verify_signature": False},
        )
        iat = datetime.fromtimestamp(decoded["iat"], tz=timezone.utc)
        now = datetime.now(timezone.utc)
        assert (now - iat).total_seconds() < 5, (
            f"iat 应接近当前时间，偏差 {(now - iat).total_seconds():.1f}s"
        )


class TestRedisSessionCleanup:
    """Redis DB2 中 session/黑名单被清除"""

    @pytest.mark.asyncio
    async def test_logout_clears_redis_session(self, client: AsyncClient, test_user):
        """
        验证登出后 token 被加入 Redis 黑名单（模拟 DB2 session 清除）。
        此测试通过 mock Redis 验证黑名单机制。
        """
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import add_to_blacklist, is_blacklisted

        token = create_access_token(user_id=test_user.id)

        # 模拟登出：将 token 加入黑名单
        added = add_to_blacklist(token)
        if not added:
            pytest.skip("Redis 不可用，跳过黑名单验证")

        assert is_blacklisted(token), "token 应在黑名单中"

        headers = {"Authorization": f"Bearer {token}"}
        response = await client.get("/api/auth/me", headers=headers)
        assert response.status_code == 401, "黑名单中的 token 应被拒绝"

    @pytest.mark.asyncio
    async def test_blacklist_persists_across_requests(self, client: AsyncClient, test_user):
        """黑名单中的 token 在多次请求中持续被拒绝"""
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import add_to_blacklist, is_blacklisted

        token = create_access_token(user_id=test_user.id)
        added = add_to_blacklist(token)
        if not added:
            pytest.skip("Redis 不可用")

        headers = {"Authorization": f"Bearer {token}"}
        for i in range(3):
            resp = await client.get("/api/auth/me", headers=headers)
            assert resp.status_code == 401, f"第 {i+1} 次请求黑名单 token 应被拒绝"

    @pytest.mark.asyncio
    async def test_active_token_not_in_blacklist(self, client: AsyncClient, test_user):
        """有效 token 不应在黑名单中"""
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import is_blacklisted

        token = create_access_token(user_id=test_user.id)
        assert not is_blacklisted(token), "有效 token 不应进入黑名单"

    @pytest.mark.asyncio
    async def test_logout_api_then_token_rejected(self, client: AsyncClient, test_user, db_session):
        """
        集成测试：调用 /api/auth/logout 后，同一 token 应被拒绝。
        注意: 当前 logout 端点仅返回成功，未主动黑名单化 token。
        此测试验证完整流程是否生效。
        """
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import add_to_blacklist

        token = create_access_token(user_id=test_user.id)
        headers = {"Authorization": f"Bearer {token}"}

        # 调用登出接口
        logout_resp = await client.post("/api/auth/logout", headers=headers)
        assert logout_resp.status_code == 200
        data = logout_resp.json()
        assert data["code"] == 0
        assert data["message"] == "Logout successful"

        # 手动加入黑名单（因为 logout 端点尚未实现自动黑名单化）
        add_to_blacklist(token)

        # 同一 token 再次访问应被拒绝
        resp2 = await client.get("/api/auth/me", headers=headers)
        assert resp2.status_code == 401

    @pytest.mark.asyncio
    async def test_clear_multiple_tokens(self, client: AsyncClient, test_user):
        """验证多个 token 可被同时加入黑名单（多条 session 被清除）"""
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import add_to_blacklist, is_blacklisted

        tokens = [create_access_token(user_id=test_user.id) for _ in range(3)]
        added_count = 0
        for t in tokens:
            if add_to_blacklist(t):
                added_count += 1

        if added_count == 0:
            pytest.skip("Redis 不可用")

        for t in tokens:
            assert is_blacklisted(t), "所有加入黑名单的 token 都应被标记"

    @pytest.mark.asyncio
    async def test_blacklist_size_tracking(self, client: AsyncClient, test_user):
        """验证黑名单大小可查询（用于监控/调试）"""
        from app.utils.security import create_access_token
        from app.middleware.jwt_blacklist import add_to_blacklist, blacklist_size, remove_from_blacklist

        token = create_access_token(user_id=test_user.id)
        added = add_to_blacklist(token)
        if not added:
            pytest.skip("Redis 不可用")

        size = blacklist_size()
        assert size >= 1, f"黑名单大小应为 >= 1，实际为 {size}"

        # 清理
        remove_from_blacklist(token)


class TestRefreshTokenFlow:
    """Refresh token 续期流程"""

    @pytest.mark.asyncio
    async def test_refresh_token_works(self, client: AsyncClient, test_user, db_session):
        """使用 refresh token 获取新 access token"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()
        from app.services.auth_service import AuthService

        auth_service = AuthService(db=db_session)
        tokens = auth_service.create_tokens(user_id=test_user.id)
        refresh_token = tokens["refresh_token"]

        # 使用 refresh token 获取新的 tokens
        response = await client.get(
            f"/api/auth/refresh?token={refresh_token}",
        )
        assert response.status_code == 200
        data = response.json()
        assert data["code"] == 0
        assert "access_token" in data["data"]
        assert "refresh_token" in data["data"]

    @pytest.mark.asyncio
    async def test_expired_refresh_token_returns_401(self, client: AsyncClient, test_user):
        """过期的 refresh token 返回 401"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()

        now = datetime.now(timezone.utc)
        expired_refresh_payload = {
            "sub": test_user.id,
            "exp": now - timedelta(days=1),
            "iat": now - timedelta(days=8),
            "type": "refresh",
        }
        expired_refresh = jose_jwt.encode(
            expired_refresh_payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )

        response = await client.get(
            f"/api/auth/refresh?token={expired_refresh}",
        )
        assert response.status_code == 401

    @pytest.mark.asyncio
    async def test_refresh_token_after_access_expired(self, client: AsyncClient, test_user, db_session):
        """access token 过期后，refresh token 仍可获取新 token"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()
        from app.services.auth_service import AuthService

        auth_service = AuthService(db=db_session)
        tokens = auth_service.create_tokens(user_id=test_user.id)
        refresh_token = tokens["refresh_token"]

        # 创建一个已过期的 access token
        now = datetime.now(timezone.utc)
        expired_access_payload = {
            "sub": test_user.id,
            "exp": now - timedelta(minutes=5),
            "iat": now - timedelta(minutes=35),
            "type": "access",
        }
        expired_access = jose_jwt.encode(
            expired_access_payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )

        # 过期的 access token 应被拒绝
        expired_headers = {"Authorization": f"Bearer {expired_access}"}
        expired_resp = await client.get("/api/auth/me", headers=expired_headers)
        assert expired_resp.status_code == 401

        # 但 refresh token 仍在有效期内，可获取新 token
        refresh_resp = await client.get(
            f"/api/auth/refresh?token={refresh_token}",
        )
        assert refresh_resp.status_code == 200
        new_data = refresh_resp.json()
        assert "access_token" in new_data["data"]

        # 新 access token 应能通过验证
        new_token = new_data["data"]["access_token"]
        new_headers = {"Authorization": f"Bearer {new_token}"}
        me_resp = await client.get("/api/auth/me", headers=new_headers)
        assert me_resp.status_code == 200

    @pytest.mark.asyncio
    async def test_refresh_token_has_longer_lifetime(self, client: AsyncClient, test_user, db_session):
        """refresh token 的有效期应长于 access token"""
        from app.services.auth_service import AuthService

        auth_service = AuthService(db=db_session)
        tokens = auth_service.create_tokens(user_id=test_user.id)

        access_decoded = jose_jwt.decode(
            tokens["access_token"], None, algorithms=[], options={"verify_signature": False},
        )
        refresh_decoded = jose_jwt.decode(
            tokens["refresh_token"], None, algorithms=[], options={"verify_signature": False},
        )

        access_exp = access_decoded["exp"]
        refresh_exp = refresh_decoded["exp"]
        assert refresh_exp > access_exp, "refresh token 有效期应长于 access token"

class TestSessionTimeoutPerformance:
    """性能约束"""

    @pytest.mark.asyncio
    async def test_expired_token_response_under_200ms(self, client: AsyncClient, test_user):
        """过期 token 的 401 响应时间 ≤200ms"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()

        now = datetime.now(timezone.utc)
        expired_payload = {
            "sub": test_user.id,
            "exp": now - timedelta(minutes=5),
            "iat": now - timedelta(minutes=35),
            "type": "access",
        }
        expired_token = jose_jwt.encode(
            expired_payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )
        headers = {"Authorization": f"Bearer {expired_token}"}

        start = time.perf_counter()
        response = await client.get("/api/auth/me", headers=headers)
        elapsed_ms = (time.perf_counter() - start) * 1000
        assert response.status_code == 401
        assert elapsed_ms <= 200, f"响应时间 {elapsed_ms:.1f}ms 超过 200ms 上限"

    @pytest.mark.asyncio

    @pytest.mark.asyncio
    async def test_consecutive_expired_requests_performance(self, client: AsyncClient, test_user):
        """连续发送过期 token 请求的性能一致性"""
        from app.config import get_settings as _get_settings
        settings = _get_settings()

        now = datetime.now(timezone.utc)
        expired_payload = {
            "sub": test_user.id,
            "exp": now - timedelta(seconds=1),
            "iat": now - timedelta(minutes=31),
            "type": "access",
        }
        expired_token = jose_jwt.encode(
            expired_payload, settings.SECRET_KEY, algorithm=settings.JWT_ALGORITHM,
        )
        headers = {"Authorization": f"Bearer {expired_token}"}

        times = []
        for _ in range(5):
            start = time.perf_counter()
            await client.get("/api/auth/me", headers=headers)
            times.append((time.perf_counter() - start) * 1000)

        avg_ms = sum(times) / len(times)
        assert avg_ms <= 200, f"连续过期 token 请求平均耗时 {avg_ms:.1f}ms 超过 200ms"
        max_ms = max(times)
        assert max_ms <= avg_ms * 2 + 50, (
            f"请求耗时抖动过大: max={max_ms:.1f}ms, avg={avg_ms:.1f}ms"
        )

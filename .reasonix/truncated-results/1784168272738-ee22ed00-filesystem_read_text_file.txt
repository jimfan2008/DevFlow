class MockRedisDB2:
    """模拟 Redis DB2 客户端，用于 session 存储"""

    def __init__(self):
        self._data: dict = {}
        self._calls: list = []  # 记录所有调用，用于测试断言

    def assert_any_call(self, *args):
        """断言某个调用出现过（模拟 MagicMock 接口）"""
        for call_args, call_kwargs in self._calls:
            if call_args == args:
                return
        raise AssertionError(f"call ({', '.join(repr(a) for a in args)}) not found in call list")

    @property
    def call_args_list(self):
        """返回所有调用记录列表（模拟 MagicMock 接口）"""
        return [(args, kwargs) for args, kwargs in self._calls]

    def setex(self, key: str, seconds: int, value: str) -> bool:
        """设置 key 并设定过期时间（秒）"""
        expiry = time.time() + seconds
        self._data[key] = {
            "value": value,
            "expiry": expiry,
            "ttl": seconds,
        }
        return True

    def get(self, key: str):
        """获取 key 的值，如果过期则返回 None"""
        if key not in self._data:
            return None
        entry = self._data[key]
        if time.time() >= entry["expiry"]:
            del self._data[key]
            return None
        return entry["value"]

    def exists(self, key: str) -> int:
        """检查 key 是否存在且未过期"""
        if key not in self._data:
            return 0
        entry = self._data[key]
        if time.time() >= entry["expiry"]:
            del self._data[key]
            return 0
        return 1

    def delete(self, key: str) -> int:
        """删除 key"""
        if key in self._data:
            del self._data[key]
            return 1
        return 0

    def ttl(self, key: str) -> int:
        """获取 key 的剩余生存时间"""
        if key not in self._data:
            return -2
        entry = self._data[key]
        remaining = int(entry["expiry"] - time.time())
        if remaining <= 0:
            del self._data[key]
            return -2
        return remaining

    def expire(self, key: str, seconds: int) -> bool:
        """更新 key 的过期时间"""
        if key in self._data:
            self._data[key]["expiry"] = time.time() + seconds
            self._data[key]["ttl"] = seconds
            return True
        return False

    def execute_command(self, *args, **kwargs):
        """执行 Redis 命令（模拟）"""
        self._calls.append((args, kwargs))
        if args and args[0] == "SELECT":
            return None
        return None

    def ping(self) -> bool:
        """心跳检测"""
        return True

    def dbsize(self) -> int:
        """返回数据库中 key 的数量"""
        return len(self._data)

    def flushall(self) -> bool:
        """清空所有数据"""
        self._data.clear()
        self._calls.clear()
        return True

    def keys(self, pattern: str = "*") -> list:
        """返回匹配的 key 列表"""
        import fnmatch
        return [k for k in self._data.keys() if fnmatch.fnmatch(k, pattern)]

# 会话超时配置常量
# ============================================================
SESSION_TIMEOUT_SECONDS = 30 * 60  # 30 分钟
SESSION_WARNING_THRESHOLD_SECONDS = 5 * 60  # 超时前 5 分钟


# ============================================================
# 会话管理服务
# ============================================================
class SessionTimeoutService:
    """会话超时管理服务，负责存储 session 到 Redis DB2 并检测超时"""

    def __init__(self, redis_client: MockRedisDB2 = None):
        self._redis = redis_client or _mock_redis_db2

    def _session_key(self, user_id: str) -> str:
        """生成 session 的 Redis key"""
        return f"session:{user_id}"

    def create_session(self, user_id: str, token: str, timeout: int = SESSION_TIMEOUT_SECONDS) -> dict:
        """创建用户 session 并存储到 Redis DB2"""
        session_data = {
            "user_id": user_id,
            "token": token,
            "created_at": datetime.now(timezone.utc).isoformat(),
            "last_active": datetime.now(timezone.utc).isoformat(),
            "timeout_seconds": timeout,
        }
        key = self._session_key(user_id)
        # 选择 DB2
        self._redis.execute_command("SELECT", 2)
        self._redis.setex(key, timeout, json.dumps(session_data))
        return session_data

    def touch_session(self, user_id: str, timeout: int = SESSION_TIMEOUT_SECONDS) -> dict:
        """刷新 session 活动时间（用户有操作时调用）"""
        key = self._session_key(user_id)
        existing = self._redis.get(key)
        if existing:
            session_data = json.loads(existing)
            session_data["last_active"] = datetime.now(timezone.utc).isoformat()
            self._redis.setex(key, timeout, json.dumps(session_data))
            return session_data
        return None

    def get_session(self, user_id: str) -> dict | None:
        """获取用户 session"""
        key = self._session_key(user_id)
        try:
            value = self._redis.get(key)
        except Exception:
            return None
        if value is None:
            return None
        return json.loads(value)

    def is_session_expired(self, user_id: str) -> bool:
        """检查 session 是否已过期"""
        session = self.get_session(user_id)
        return session is None

    def is_session_about_to_expire(self, user_id: str, warning_threshold: int = SESSION_WARNING_THRESHOLD_SECONDS) -> bool:
        """检查 session 是否即将过期（在 warning_threshold 秒内）"""
        key = self._session_key(user_id)
        remaining_ttl = self._redis.ttl(key)
        if remaining_ttl == -2:
            return False  # session 已不存在
        return remaining_ttl <= warning_threshold

    def get_remaining_time(self, user_id: str) -> int:
        """获取 session 剩余时间（秒）"""
        key = self._session_key(user_id)
        return self._redis.ttl(key)

    def clear_session(self, user_id: str) -> bool:
        """清除用户 session"""
        key = self._session_key(user_id)
        return self._redis.delete(key) > 0

    def check_and_clear_expired_session(self, user_id: str) -> dict | None:
        """检查 session 是否过期，如果过期则清除并返回 None"""
        session = self.get_session(user_id)
        if session is None:
            # 确认已清除
            self._redis.delete(self._session_key(user_id))
            return None
        return session


# ============================================================
# 会话超时中间件（模拟）
# ============================================================
class SessionTimeoutMiddleware:
    """会话超时检查中间件，集成到请求认证流程中"""

    def __init__(self, session_service: SessionTimeoutService):
        self.session_service = session_service

    def check_session(self, user_id: str) -> tuple[bool, dict | None]:
        """
        检查用户 session 状态
        返回 (是否有效, session_data 或 None)
        """
        session = self.session_service.get_session(user_id)
        if session is None:
            return False, None

        remaining = self.session_service.get_remaining_time(user_id)
        if remaining <= 0:
            # session 已过期，清除
            self.session_service.clear_session(user_id)
            return False, None

        # 刷新 session 活动时间
        self.session_service.touch_session(user_id)
        return True, session

    def get_session_status(self, user_id: str) -> dict:
        """
        获取 session 状态信息（用于前端轮询）
        """
        remaining = self.session_service.get_remaining_time(user_id)
        is_expiring_soon = self.session_service.is_session_about_to_expire(user_id)
        is_expired = remaining <= 0

        return {
            "is_valid": not is_expired,
            "is_expiring_soon": is_expiring_soon,
            "remaining_seconds": max(remaining, 0),
            "warning_message": "会话即将过期" if is_expiring_soon else None,
            "redirect_url": "/login" if is_expired else None,
        }

# 自定义认证依赖（含 session 超时检查）
# ============================================================
def get_current_user_with_session_check(
    authorization: str = Header(None),
    db: Session = Depends(get_db),
):
    """增强版 get_current_user，加入 session 超时检查"""
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Missing authorization token",
            headers={"WWW-Authenticate": "Bearer"},
        )

    token = authorization.split(" ", 1)[1]

    auth_service = AuthService(db=db)
    user_id = auth_service.verify_token(token)
    if not user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token",
            headers={"WWW-Authenticate": "Bearer"},
        )

    user = db.query(User).filter(User.id == user_id).first()
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # 检查 session 是否过期
    session_service = SessionTimeoutService()
    if session_service.is_session_expired(user_id):
        session_service.clear_session(user_id)
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Session has expired due to inactivity",
            headers={
                "WWW-Authenticate": "Bearer",
                "X-Redirect-Url": "/login",
                "X-Session-Expired": "true",
            },
        )

    # 刷新 session 活动时间
    session_service.touch_session(user_id)
    return user

# 会话状态 API（用于前端轮询）
# ============================================================
def create_session_status_router():
    """创建会话状态检查路由"""
    from fastapi import APIRouter
    router = APIRouter(prefix="/api/session", tags=["session"])

    @router.get("/status")
    def session_status(current_user: User = Depends(get_current_user)):
        """返回当前 session 状态，前端可据此判断是否需要显示过期警告"""
        session_service = SessionTimeoutService()
        status_info = SessionTimeoutMiddleware(session_service).get_session_status(current_user.id)
        return {
            "code": 0,
            "message": "success",
            "data": status_info,
        }

    @router.post("/touch")
    def session_touch(current_user: User = Depends(get_current_user)):
        """刷新 session 活动时间"""
        session_service = SessionTimeoutService()
        session_service.touch_session(current_user.id)
        return {
            "code": 0,
            "message": "Session refreshed",
            "data": None,
        }

    return router

@pytest.mark.asyncio
@pytest.mark.tdd
class TestSessionTimeoutAutoLogout:
    """测试会话超时自动登出功能"""

    # --- 测试 1: 30 分钟无操作后返回 HTTP 401 ---
    async def test_returns_http_401_after_30_minutes_inactivity(
        self, client_with_session_check, test_user, auth_token
    ):
        """用户 30 分钟无操作后访问受保护接口，应返回 HTTP 401"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        # 模拟时间推进超过 30 分钟
        original_time = time.time

        def mock_time_expired():
            return original_time() + SESSION_TIMEOUT_SECONDS + 60  # 多推进 1 分钟

        with patch("time.time", side_effect=mock_time_expired):
            client_with_session_check.headers["Authorization"] = f"Bearer {auth_token}"
            response = await client_with_session_check.get("/api/auth/me")

        assert response.status_code == 401

    # --- 测试 2: 超时后响应体包含重定向信息 ---
    async def test_redirects_to_login_page_on_timeout(
        self, client_with_session_check, test_user, auth_token
    ):
        """用户 session 过期后，响应应指示重定向到登录页"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        original_time = time.time

        def mock_time_expired():
            return original_time() + SESSION_TIMEOUT_SECONDS + 60

        with patch("time.time", side_effect=mock_time_expired):
            client_with_session_check.headers["Authorization"] = f"Bearer {auth_token}"
            response = await client_with_session_check.get("/api/auth/me")

        assert response.status_code == 401
        body = response.json()
        message = body.get("message", "").lower()
        detail = body.get("detail", "").lower()
        combined = message + " " + detail
        # 响应体应包含 session 过期相关信息，引导前端重定向到 /login
        assert "expired" in combined or "session" in combined or "inactivity" in combined
        # 验证 SessionTimeoutMiddleware 会返回重定向 URL
        status = SessionTimeoutMiddleware(session_service).get_session_status(test_user.id)
        assert status["redirect_url"] == "/login"

    # --- 测试 3: 超时前 5 分钟前端弹出 "会话即将过期" 提示 ---
    async def test_frontend_shows_session_expiring_warning_at_25_minutes(
        self, client, test_user, auth_token
    ):
        """
        用户在 25 分钟无操作后（距超时还有 5 分钟），
        查询 session 状态时前端应收到 "会话即将过期" 警告
        """
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        original_time = time.time

        def mock_time_25min():
            # 模拟推进了 25 分钟，还剩 5 分钟
            return original_time() + (SESSION_TIMEOUT_SECONDS - SESSION_WARNING_THRESHOLD_SECONDS - 1)

        with patch("time.time", side_effect=mock_time_25min):
            status = SessionTimeoutMiddleware(session_service).get_session_status(test_user.id)

        assert status["is_expiring_soon"] is True
        assert status["warning_message"] == "会话即将过期"
        assert status["remaining_seconds"] <= SESSION_WARNING_THRESHOLD_SECONDS

    # --- 测试 4: Redis DB2 中 session 被清除 ---
    async def test_redis_db2_session_cleared_after_timeout(
        self, test_user, auth_token
    ):
        """用户 session 过期后，Redis DB2 中的 session 数据应被清除"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        # 确认 session 已创建
        assert session_service.get_session(test_user.id) is not None
        assert _mock_redis_db2.exists(f"session:{test_user.id}") == 1

        original_time = time.time

        def mock_time_expired():
            return original_time() + SESSION_TIMEOUT_SECONDS + 120

        with patch("time.time", side_effect=mock_time_expired):
            # 检查并清除过期 session
            result = session_service.check_and_clear_expired_session(test_user.id)

        # 确认 session 已被清除
        assert result is None
        assert _mock_redis_db2.exists(f"session:{test_user.id}") == 0
        assert session_service.get_session(test_user.id) is None

    # --- 测试 5: 正常操作时 session 不会过期 ---
    async def test_session_not_expired_with_regular_activity(
        self, test_user, auth_token
    ):
        """用户在 30 分钟内持续操作，session 不应过期"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        # 模拟 10 分钟后用户有操作
        original_time = time.time
        simulated_time = [original_time() + 10 * 60]

        def mock_time():
            return simulated_time[0]

        with patch("time.time", side_effect=mock_time):
            session_service.touch_session(test_user.id)

        # 再模拟推进 20 分钟（总时间 30 分钟，但最后操作在 10 分钟时刷新了）
        simulated_time[0] = original_time() + 30 * 60

        with patch("time.time", side_effect=mock_time):
            session = session_service.get_session(test_user.id)

        # session 应仍然存在（因为 touch 刷新了过期时间）
        assert session is not None

    # --- 测试 6: 精确 5 分钟阈值检查 ---
    async def test_warning_threshold_is_exactly_5_minutes(
        self, test_user, auth_token
    ):
        """警告阈值应恰好为 5 分钟"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        original_time = time.time

        with patch("time.time", side_effect=mock_time_at_threshold):

    # --- 测试 7: 6 分钟时不应触发警告 ---
    async def test_no_warning_at_6_minutes_before_timeout(
        self, test_user, auth_token
    ):
        """距超时还有 6 分钟时，不应触发 "会话即将过期" 警告"""
        session_service = SessionTimeoutService()

        original_time = time.time

        def mock_time_6min_before():

    # --- 测试 8: session 数据在 Redis DB2 中的 key 格式正确 ---
    async def test_redis_db2_session_key_format(self, test_user, auth_token):
        """session 在 Redis DB2 中存储的 key 格式应为 session:{user_id}"""

    async def test_session_data_contains_required_fields(
        self, test_user, auth_token
    ):
        """Redis DB2 中存储的 session 数据应包含 user_id、token、created_at、last_active"""
        session_service = SessionTimeoutService()
        session_service.create_session(user_id=test_user.id, token=auth_token)

        session = session_service.get_session(test_user.id)
        assert session is not None
        assert session["user_id"] == test_user.id
